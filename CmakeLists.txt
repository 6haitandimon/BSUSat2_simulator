cmake_minimum_required(VERSION 3.13)


set(PICO_SDK_PATH "${CMAKE_SOURCE_DIR}/external/pico-sdk")

#add sdk import
include(pico_sdk_import.cmake)

project(CubeBEL_2_SIM C CXX ASM)

pico_sdk_init()

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(DEVELOP_MODE 1)

if ("${CMAKE_BUILD_TYPE}" STREQUAL "Release")
    message(STATUS "Maximum optimization for speed")
    add_compile_options(-Ofast)
elseif ("${CMAKE_BUILD_TYPE}" STREQUAL "RelWithDebInfo")
    message(STATUS "Maximum optimization for speed, debug info included")
    add_compile_options(-Ofast -g)
elseif ("${CMAKE_BUILD_TYPE}" STREQUAL "MinSizeRel")
    message(STATUS "Maximum optimization for size")
    add_compile_options(-Os)
elseif ("${CMAKE_BUILD_TYPE}" STREQUAL "Debug")
    message(STATUS "Max debug info")
    add_compile_options(-Og -g)
else ()
    message(STATUS "Minimal optimization, debug info included")
    add_compile_options(-Og -g)
endif ()

#add libs
add_subdirectory(libs)

#temporary
add_subdirectory(INTERRUPT)


if (DEVELOP_MODE STREQUAL 0)
    file(GLOB MODULES RELATIVE ${CMAKE_SOURCE_DIR}/modules modules/*)

    list(LENGTH MODULES MODULE_LIST_SIZE)

    set(MODULE_TO_BUILD "" CACHE STRING "Enter number to select module")

    if(MODULE_TO_BUILD LESS ${MODULE_LIST_SIZE})
        list(GET MODULES ${MODULE_TO_BUILD} SELECTED_MODULE)
        message(STATUS "Select module: ${SELECTED_MODULE}")
        add_subdirectory(modules/${SELECTED_MODULE})
    else()
        message(WARNING "Invalid selection, no module selected.")
    endif()
else ()
    #manual add modules from develop
    add_subdirectory(modules/MotherBoard)
endif ()
